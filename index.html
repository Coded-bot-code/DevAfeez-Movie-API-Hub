Operational 
// Search for a movie
const res = await fetch(
  "https://movieapi.devafeez.name.ng/api/v1/search/avengers",
  { headers: { Authorization: "Bearer YourApiKey" } }
);
const data = await res.json();
// data.results.items → array of matching titles

// Get streaming sources for a title
const sources = await fetch(
  `https://movieapi.devafeez.name.ng/api/v1/sources/${data.results.items[0].subjectId}`,
  { headers: { Authorization: "Bearer YourApiKey" } }
);
const { results } = await sources.json();
// results[0].stream_url  → proxied HLS/MP4 stream URL
// results[0].quality     → "1080p" | "720p" | "480p"
# Search for a movie
curl -X GET "https://movieapi.devafeez.name.ng/api/v1/search/avengers" \
     -H "Authorization: Bearer YourApiKey"

# Get streaming sources
curl -X GET "https://movieapi.devafeez.name.ng/api/v1/sources/{id}" \
     -H "Authorization: Bearer YourApiKey"

# Trending titles
curl -X GET "https://movieapi.devafeez.name.ng/api/v1/trending" \
     -H "Authorization: Bearer YourApiKey"
import requests

API_KEY = "YourApiKey"
BASE    = "https://movieapi.devafeez.name.ng/api/v1"
HEADERS = {"Authorization": f"Bearer {API_KEY}"}

# Search for a movie
res  = requests.get(f"{BASE}/search/avengers", headers=HEADERS).json()
item = res["results"]["items"][0]

# Get streaming sources
src  = requests.get(f"{BASE}/sources/{item['id']}", headers=HEADERS).json()
url  = src["results"]["sources"][0]["url"]
